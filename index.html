async function loadModel(retries = 3) { for (let attempt = 1; attempt <= retries; attempt++) { try { statusText.textContent = attempt === 1 ? "Fetching model.json..." : `Retrying model.json (attempt ${attempt}/${retries})...`; const response = await fetch(`./model.json?t=${Date.now()}`, { cache: "no-store", }); if (!response.ok) { throw new Error(`HTTP ${response.status}`); } cropModel = await response.json(); predictBtn.disabled = false; predictBtn.textContent = "Recommend Crop"; statusText.textContent = `Model ready (${cropModel.classes.length} crops)`; return; } catch (err) { if (attempt === retries) { statusText.textContent = "Error loading model.json: " + err.message; predictBtn.textContent = "Model Unavailable"; } else { await new Promise((r) => setTimeout(r, 1200 * attempt)); } } } }
